Private Function Min2(ByVal a As Double, ByVal b As Double) As Double
    If a < b Then Min2 = a Else Min2 = b
End Function

Private Function Max2(ByVal a As Double, ByVal b As Double) As Double
    If a > b Then Max2 = a Else Max2 = b
End Function

' Yelim qanoti: x chizig'iga chapdan ulangan, ~15 gradus qiya
Private Sub Glue(ByVal x As Double, ByVal H As Double, ByVal g As Double)
    Dim c As Double
    c = Min2(g * 0.3, H / 4)
    Poly x, 0, x - g, c, x - g, H - c, x, H
End Sub

' Yelim | w0 | w1 | ...  yon devorlar (balandligi H). xs - panel chegaralari
Private Sub Body(xs() As Double, ByVal H As Double, ByVal g As Double, ParamArray widths() As Variant)
    Dim i As Long, n As Long
    n = UBound(widths) + 1
    ReDim xs(0 To n)
    For i = 0 To n - 1: xs(i + 1) = xs(i) + widths(i): Next i
    Glue 0, H, g
    For i = 0 To n - 1: Rect xs(i), 0, xs(i + 1), H: Next i
End Sub

' ---- nuqtalar ro'yxati (yumaloq burchakli shakllar uchun)
Private Sub PtBegin()
    ptN = 0
    ReDim ptX(0 To 63): ReDim ptY(0 To 63)
End Sub

Private Sub Pt(ByVal x As Double, ByVal y As Double)
    If ptN > UBound(ptX) Then
        ReDim Preserve ptX(0 To UBound(ptX) * 2 + 1): ReDim Preserve ptY(0 To UBound(ptY) * 2 + 1)
    End If
    ptX(ptN) = x: ptY(ptN) = y: ptN = ptN + 1
End Sub

' Yoy nuqtalari (gradus), uchlari bilan
Private Sub ArcPt(ByVal cx As Double, ByVal cy As Double, ByVal rad As Double, ByVal a0 As Double, ByVal a1 As Double)
    Dim i As Long, a As Double
    For i = 0 To 6
        a = (a0 + (a1 - a0) * i / 6) * PI / 180
        Pt cx + rad * Cos(a), cy + rad * Sin(a)
    Next i
End Sub

' Ellips yoyi (gradus), uchlari bilan
Private Sub EllPt(ByVal cx As Double, ByVal cy As Double, ByVal rx As Double, ByVal ry As Double, _
                  ByVal a0 As Double, ByVal a1 As Double, ByVal n As Long)
    Dim i As Long, a As Double
    For i = 0 To n
        a = (a0 + (a1 - a0) * i / n) * PI / 180
        Pt cx + rx * Cos(a), cy + ry * Sin(a)
    Next i
End Sub

Private Sub PtPoly()
    PolyArr ptX, ptY, ptN
End Sub

Private Sub PtCut()
    CutPolyArr ptX, ptY, ptN
End Sub

' Yumaloq burchakli to'rtburchak teshik (oyna)
Private Sub CutRoundRect(ByVal x1 As Double, ByVal y1 As Double, ByVal x2 As Double, ByVal y2 As Double, ByVal r As Double)
    r = Min2(r, Min2(x2 - x1, y2 - y1) / 2)
    PtBegin
    ArcPt x2 - r, y1 + r, r, 270, 360
    ArcPt x2 - r, y2 - r, r, 0, 90
    ArcPt x1 + r, y2 - r, r, 90, 180
    ArcPt x1 + r, y1 + r, r, 180, 270
    PtCut
End Sub

' Ruchka teshigi (stadion shakli): markaz, eni, balandligi
Private Sub CutStadium(ByVal cx As Double, ByVal cy As Double, ByVal w As Double, ByVal h As Double)
    Dim r As Double
    r = h / 2
    If w < h Then w = h
    PtBegin
    ArcPt cx + w / 2 - r, cy, r, 270, 450
    ArcPt cx - w / 2 + r, cy, r, 90, 270
    PtCut
End Sub

' Evro osma teshigi: gorizontal tirqish (balandligi 5), o'rtasida dumaloq (radiusi 6)
Private Sub CutEuroSlot(ByVal cx As Double, ByVal cy As Double, ByVal w As Double)
    Dim h As Double, r As Double, a As Double, q As Double
    h = 5: r = 6
    q = h / 2 / r: a = Atn(q / Sqr(1 - q * q)) * 180 / PI     ' Asin
    PtBegin
    Pt cx - w / 2, cy - h / 2
    ArcPt cx, cy, r, 180 + a, 360 - a
    Pt cx + w / 2, cy - h / 2: Pt cx + w / 2, cy + h / 2
    ArcPt cx, cy, r, a, 180 - a
    Pt cx - w / 2, cy + h / 2
    PtCut
End Sub

' Ikki qavat devorning ichki qismi qulf tillari bilan. x0 - devor bukish chizig'i, sx - yo'nalish,
' W - devor uzunligi. swapXY = True bo'lsa o'qlar almashadi (old/orqa devor uchun)
Private Sub LockWall(ByVal x0 As Double, ByVal sx As Double, ByVal W As Double, ByVal H As Double, _
                     ByVal t As Double, ByVal swapXY As Boolean)
    Dim xi As Double, xe As Double, td As Double, k As Long, a As Double, b As Double
    xi = x0 + sx * H: xe = x0 + sx * (2 * H - t)
    td = Max2(2 * t, 3)
    PtBegin
    PtS xi, 0, swapXY: PtS xe, 0, swapXY
    For k = 0 To 1
        If k = 0 Then a = 0.2 * W: b = 0.4 * W Else a = 0.6 * W: b = 0.8 * W
        PtS xe, a, swapXY: PtS xe + sx * td, a + 1, swapXY
        PtS xe + sx * td, b - 1, swapXY: PtS xe, b, swapXY
    Next k
    PtS xe, W, swapXY: PtS xi, W, swapXY
    PtPoly
End Sub

Private Sub PtS(ByVal x As Double, ByVal y As Double, ByVal swapXY As Boolean)
    If swapXY Then Pt y, x Else Pt x, y
End Sub

' Taglik bilan yon devor orasidagi bukishda qulf tillari uchun tirqishlar
Private Sub BaseSlots(ByVal x As Double, ByVal W As Double, ByVal swapXY As Boolean)
    If swapXY Then
        Slit 0.2 * W, x, 0.4 * W, x: Slit 0.6 * W, x, 0.8 * W, x
    Else
        Slit x, 0.2 * W, x, 0.4 * W: Slit x, 0.6 * W, x, 0.8 * W
    End If
End Sub

' FEFCO qanotlari, qo'shni qanotlar orasida tirqish
Private Sub SlotFlaps(xs() As Double, ByVal baseY As Double, ByVal s As Double, ByVal f As Double)
    Dim i As Long
    For i = 0 To UBound(xs) - 1: Rect xs(i), baseY, xs(i + 1), baseY + s * f: Next i
    For i = 1 To UBound(xs) - 1: Slit xs(i), baseY, xs(i), baseY + s * f: Next i
End Sub

' Piramida yoki tomning qo'shni yuzalari orasidagi burilish burchagi (gradus): yelim qanoti shuncha buklanadi.
' n - tomonlar soni, rin - taglik ichki radiusi, h - balandlik
Private Function FlapAng(ByVal n As Long, ByVal rin As Double, ByVal h As Double) As Double
    Dim c As Double
    c = (h * h * Cos(2 * PI / n) + rin * rin) / (h * h + rin * rin)
    If c >= 1 Then FlapAng = 0: Exit Function
    FlapAng = (Atn(-c / Sqr(1 - c * c)) + PI / 2) * 180 / PI      ' Acos
End Function

' Muntazam ko'pburchak: birinchi qirrasi (x1,y1)-(x2,y2), soat strelkasiga teskari. Nuqtalar ptX/ptY da
Private Sub RegPoly(ByVal x1 As Double, ByVal y1 As Double, ByVal x2 As Double, ByVal y2 As Double, ByVal n As Long)
    Dim k As Long, ex As Double, ey As Double, a As Double, c As Double, s As Double, x As Double, y As Double
    ex = x2 - x1: ey = y2 - y1
    PtBegin
    x = x1: y = y1
    For k = 0 To n - 1
        Pt x, y
        a = 2 * PI * k / n: c = Cos(a): s = Sin(a)
        x = x + ex * c - ey * s: y = y + ex * s + ey * c
    Next k
End Sub

' Arccos (VBA da yo'q)
Private Function Acs(ByVal c As Double) As Double
    If c >= 1 Then
        Acs = 0
    ElseIf c <= -1 Then
        Acs = PI
    Else
        Acs = Atn(-c / Sqr(1 - c * c)) + PI / 2
    End If
End Function

' Arcsin
Private Function Asn(ByVal s As Double) As Double
    Asn = PI / 2 - Acs(s)
End Function

' Atan2: (x, y) vektorining burchagi (radian)
Private Function Ang2(ByVal y As Double, ByVal x As Double) As Double
    If x > 0 Then
        Ang2 = Atn(y / x)
    ElseIf x < 0 Then
        If y >= 0 Then Ang2 = Atn(y / x) + PI Else Ang2 = Atn(y / x) - PI
    Else
        If y >= 0 Then Ang2 = PI / 2 Else Ang2 = 0 - PI / 2
    End If
End Function

' Yumaloq burchak nuqtalari. (px, py) - to'g'ri burchak uchi, ux/uy (+1/-1) - burchak qaysi tomonga qaragan,
' alongX = True - kontur burchakka X o'qi bo'ylab keladi
Private Sub RCorner(ByVal px As Double, ByVal py As Double, ByVal ux As Double, ByVal uy As Double, _
                    ByVal r As Double, ByVal alongX As Boolean)
    Dim ax As Double, ay As Double
    If r <= 0.01 Then
        Pt px, py
        Exit Sub
    End If
    If ux > 0 Then ax = 0 Else ax = 180
    If uy > 0 Then ay = 90 Else ay = -90
    If ax - ay > 180 Then ay = ay + 360
    If ay - ax > 180 Then ay = ay - 360
    If alongX Then ArcPt px - ux * r, py - uy * r, r, ay, ax Else ArcPt px - ux * r, py - uy * r, r, ax, ay
End Sub

' To'rtburchak panel; y = yb qirrasi o'rtasida yarim doira barmoq o'yig'i (r = 0 - o'yiqsiz)
Private Sub NotchRect(ByVal xa As Double, ByVal ya As Double, ByVal xb As Double, ByVal yb As Double, ByVal r As Double)
    Dim cx As Double
    If r <= 0 Then
        Rect xa, ya, xb, yb
        Exit Sub
    End If
    cx = (xa + xb) / 2
    PtBegin
    Pt xa, ya: Pt xb, ya: Pt xb, yb
    If yb > ya Then ArcPt cx, yb, r, 0, -180 Else ArcPt cx, yb, r, 0, 180
    Pt xa, yb
    PtPoly
End Sub

' Barmoq o'yig'i radiusi (fx da TN bo'lsa)
Private Function TnR(ByVal fx As String, ByVal L As Double) As Double
    If Has(fx, "TN") Then TnR = Min2(12, L * 0.07)
End Function

' Qirraga ulangan trapetsiya qanot: qirra (x1,y1) -> (x2,y2), qanot o'ng tomonda, eni g, uchlari c ga qiya
Private Sub EdgeFlap(ByVal x1 As Double, ByVal y1 As Double, ByVal x2 As Double, ByVal y2 As Double, _
                     ByVal g As Double, ByVal c As Double)
    Dim dx As Double, dy As Double, ln As Double, nx As Double, ny As Double
    dx = x2 - x1: dy = y2 - y1: ln = Sqr(dx * dx + dy * dy)
    If ln < 0.001 Then Exit Sub
    dx = dx / ln: dy = dy / ln: nx = dy: ny = 0 - dx
    c = Min2(c, ln / 3)
    Poly x1, y1, x2, y2, x2 + nx * g - dx * c, y2 + ny * g - dy * c, x1 + nx * g + dx * c, y1 + ny * g + dy * c
End Sub

' Perforatsiya (yirtiladigan chiziq): qisqa kesishlar
Private Sub Perf(ByVal x1 As Double, ByVal y1 As Double, ByVal x2 As Double, ByVal y2 As Double)
    Dim ln As Double, n As Long, k As Long, t0 As Double, t1 As Double
    ln = Sqr((x2 - x1) ^ 2 + (y2 - y1) ^ 2)
    n = Int(ln / 5)
    If n < 1 Then n = 1
    For k = 0 To n - 1
        t0 = (k + 0.2) / n: t1 = (k + 0.8) / n
        CutPoly x1 + (x2 - x1) * t0, y1 + (y2 - y1) * t0, x1 + (x2 - x1) * t1, y1 + (y2 - y1) * t1
    Next k
End Sub

' Tik (Y bo'ylab) stadion teshik: markaz, uzunligi, eni
Private Sub CutSlotV(ByVal cx As Double, ByVal cy As Double, ByVal ln As Double, ByVal th As Double)
    Dim r As Double
    r = th / 2
    If ln < th Then ln = th
    PtBegin
    ArcPt cx, cy + ln / 2 - r, r, 0, 180
    ArcPt cx, cy - ln / 2 + r, r, 180, 360
    PtCut
End Sub

' Yon qanot: x chizig'iga (ya..yb) ulangan, sx tomonga dd, tashqi burchaklari yumaloq
Private Sub SideFlapR(ByVal x As Double, ByVal ya As Double, ByVal yb As Double, ByVal sx As Double, ByVal dd As Double)
    Dim r As Double
    r = Min2(dd * 0.8, (yb - ya) * 0.45)
    PtBegin
    Pt x, ya
    RCorner x + sx * dd, ya, sx, -1, r, True
    RCorner x + sx * dd, yb, sx, 1, r, False
    Pt x, yb
    PtPoly
End Sub

' Quloq: (x0,y0) burchakdan sx tomonga eLen, sy tomonga eh. Devorga x = x0 qirrasi bilan ulangan,
' y = y0 qirrasi - tirqish. fx: RE - uchi yumaloq, HK - ilgakli (qopqoq quloqlarini ushlaydi)
Private Sub Ear(ByVal x0 As Double, ByVal y0 As Double, ByVal sx As Double, ByVal sy As Double, _
                ByVal eLen As Double, ByVal eh As Double, ByVal fx As String)
    Dim xe As Double, yt As Double
    xe = x0 + sx * eLen: yt = y0 + sy * eh
    PtBegin
    Pt x0, y0: Pt xe, y0
    If Has(fx, "RE") Or Has(fx, "HK") Then RCorner xe, yt, sx, sy, Min2(eLen, eh) * 0.45, False Else Pt xe, yt
    If Has(fx, "HK") Then
        Pt x0 + sx * eLen * 0.62, yt
        Pt x0 + sx * eLen * 0.55, yt - sy * eh * 0.35
        Pt x0 + sx * eLen * 0.47, yt - sy * eh * 0.35
        Pt x0 + sx * eLen * 0.47, yt
    End If
    Pt x0, yt
    PtPoly
    Slit x0, y0, xe, y0
End Sub

' Devor davomidagi ruchka: x - devor cheti, sx - yo'nalish, (ya..yb) bo'ylab, ht - balandligi.
' Uchi yumaloq, o'rtasida ruchka teshigi
Private Sub HandleTab(ByVal x As Double, ByVal sx As Double, ByVal ya As Double, ByVal yb As Double, ByVal ht As Double)
    Dim r As Double
    r = Min2(ht * 0.9, (yb - ya) * 0.45)
    PtBegin
    Pt x, ya
    RCorner x + sx * ht, ya, sx, -1, r, True
    RCorner x + sx * ht, yb, sx, 1, r, False
    Pt x, yb
    PtPoly
    SetFold x, ya, x, yb, 0                      ' devor davomi: tik ko'tariladi
    CutSlotV x + sx * ht * 0.5, (ya + yb) / 2, Min2((yb - ya) * 0.5, 90), Min2(22, ht * 0.4)
End Sub

' ================================================================ gofra

' st: 0 - ustki qanotsiz (0200), 1 - 0201, 3 - to'liq qopqoq (0203). rt - ruchka teshigi eni
Private Sub Rsc(ByVal L As Double, ByVal W As Double, ByVal H As Double, ByVal g As Double, _
                ByVal st As Long, ByVal rt As Double)
    Dim xs() As Double, f As Double, k As Long, hh As Double
    Body xs, H, g, L, W, L, W
    f = W / 2
    If st = 3 Then f = W
    If st <> 0 Then SlotFlaps xs, H, 1, f
    SlotFlaps xs, 0, -1, f
    If rt > 0 Then
        hh = Min2(30, rt * 0.35)
        For k = 1 To 3 Step 2
            CutStadium (xs(k) + xs(k + 1)) / 2, H - Max2(25, hh * 1.5), Min2(rt, W - 20), hh
        Next k
    End If
    SetRoot (xs(2) + xs(3)) / 2, H / 2, True       ' 3D: orqa panel tik turadi, ustki qanotlar - qopqoq
    If st <> 0 Then                              ' ichki (W) qanotlar ostida, tashqi (L) qanotlar ustida
        SetLid xs(0), H, xs(1), H, 1: SetLid xs(2), H, xs(3), H, 1
        SetLid xs(1), H, xs(2), H, 2: SetLid xs(3), H, xs(4), H, 2
    End If
    Box3 0, 0, 0, L, H, W, st <> 0
End Sub

' ================================================================ mailer

' Mailer (FEFCO 0427). dbl - yon devorlar ikki qavat (qulf tillari bilan), aks holda bir qavat.
' lf - qopqoq quloqlari, oy - oyna, il - osma teshik, nq - qulf tillari soni, rt - old devorda ruchka teshigi.
' fx: TN - old devorda barmoq o'yig'i, RE - yumaloq quloq, HK - ilgakli quloq, LR - qopqoq quloqlari yumaloq,
'     TS - yirtiladigan tasma, SH - yon devorlarda ruchka
Private Sub Mailer(ByVal L As Double, ByVal W As Double, ByVal H As Double, ByVal t As Double, _
                   ByVal lf As Boolean, ByVal oy As Double, ByVal il As Boolean, ByVal nq As Long, _
                   ByVal rt As Double, ByVal dbl As Boolean, ByVal fx As String)
    Dim eLen As Double, dd As Double, k As Double, y0 As Double, ht As Double, e As Double, rr As Double
    Dim x0 As Double, sx As Double, yy As Double, sy As Double, a As Long, b As Long, m As Double
    Dim tw As Double, tdp As Double, cx As Double
    If dbl Then eLen = Min2(1.3 * H, W / 2 - 2) Else eLen = Min2(H, W / 2 - 2)
    Rect 0, 0, L, W                              ' taglik
    NotchRect 0, 0, L, 0 - H, TnR(fx, L)          ' old devor
    Rect 0, W, L, W + H                          ' orqa devor
    For a = 0 To 1
        If a = 0 Then x0 = 0: sx = -1 Else x0 = L: sx = 1
        Rect x0, 0, x0 + sx * H, W               ' yon devor
        If dbl Then
            LockWall x0, sx, W, H, t, False      ' ichki yon devor
            BaseSlots x0, W, False
            SetFold x0 + sx * H, 0, x0 + sx * H, W, 180
        End If
        If Has(fx, "SH") Then HandleSide x0 + sx * H * 0.5, W, W * 0.5, H
        For b = 0 To 1                           ' quloqlar old/orqa devorga ulangan
            If b = 0 Then yy = 0: sy = -1 Else yy = W: sy = 1
            Ear x0, yy, sx, sy, eLen, H - t, fx
        Next b
    Next a
    Rect 0, W + H, L, 2 * W + H                  ' qopqoq
    If lf Then                                   ' qopqoq quloqlari
        dd = H - 2 * t: k = dd * 0.35
        If Has(fx, "LR") Then
            SideFlapR 0, W + H + t, 2 * W + H - t, -1, dd
            SideFlapR L, W + H + t, 2 * W + H - t, 1, dd
        Else
            Poly 0, W + H + t, 0 - dd, W + H + t + k, 0 - dd, 2 * W + H - t - k, 0, 2 * W + H - t
            Poly L, W + H + t, L + dd, W + H + t + k, L + dd, 2 * W + H - t - k, L, 2 * W + H - t
        End If
    End If
    y0 = 2 * W + H: ht = H - t                   ' old til
    tw = L * 0.12: tdp = Max2(2 * t, 4)
    If nq = 1 Then tw = L * 0.18
    PtBegin
    Pt 0, y0: Pt 0, y0 + ht
    For a = 0 To nq - 1                          ' qulf tillari va ular uchun tirqishlar
        cx = L * (a + 0.5) / nq
        Pt cx - tw / 2, y0 + ht: Pt cx - tw / 2 + 1, y0 + ht + tdp
        Pt cx + tw / 2 - 1, y0 + ht + tdp: Pt cx + tw / 2, y0 + ht
        Slit cx - tw / 2, 0, cx + tw / 2, 0
    Next a
    Pt L, y0 + ht: Pt L, y0
    PtPoly
    e = H - 2 * t: rr = Min2(e, ht) * 0.9
    PtBegin
    Pt 0, y0: Pt 0 - e, y0: ArcPt rr - e, y0 + ht - rr, rr, 180, 90: Pt 0, y0 + ht
    PtPoly
    PtBegin
    Pt L, y0: Pt L + e, y0: ArcPt L + e - rr, y0 + ht - rr, rr, 0, 90: Pt L, y0 + ht
    PtPoly
    If oy > 0 Then
        m = Min2(oy, Min2(L, W) / 2 - 5)
        CutRoundRect m, W + H + m, L - m, 2 * W + H - m, 5
    End If
    If il Then CutEuroSlot L / 2, y0 + ht / 2, Min2(L * 0.3, 40)
    If Has(fx, "HL") Then CutStadium L / 2, W + H + W * 0.5, Min2(L * 0.4, 100), Min2(25, W * 0.15)   ' qopqoqda ruchka
    If Has(fx, "TS") Then                        ' yirtiladigan tasma qopqoq bo'ylab
        Perf 0, y0 - 12, L, y0 - 12
        Perf 0, y0 - 20, L, y0 - 20
    End If
    If rt > 0 And H >= 30 Then CutStadium L / 2, 0 - H / 2, Min2(rt, L - 30), Min2(25, H * 0.4)
    SetRoot L / 2, W / 2, False                  ' 3D: taglik yotadi, qopqoq orqa devorda
    SetLid 0, W + H, L, W + H
    Box3 0, 0, 0, L, H, W, False
    Lid3 L, H, W, 70, H - t
End Sub

' ================================================================ tuck-end

' rev - teskari (pastki qopqoq qarama-qarshi panelda). bot: 0 - til, 1 - qulf tag (1-2-3), 2 - yopishtiriladigan.
' oy - old panelda oyna, il - osma qism balandligi (0 - yo'q)
Private Sub TuckEnd(ByVal L As Double, ByVal W As Double, ByVal H As Double, ByVal g As Double, _
                    ByVal tuck As Double, ByVal rev As Boolean, ByVal bot As Long, ByVal oy As Double, ByVal il As Double, _
                    ByVal fx As String)
    Dim xs() As Double, m As Double, c As Double
    Body xs, H, g, L, W, L, W
    TuckClosure xs, H, 1, 0, L, W, tuck
    Select Case bot
        Case 1: SnapBottom xs, L, W
        Case 2: SealEnd xs, L, W
        Case Else
            If rev Then TuckClosure xs, 0, -1, 2, L, W, tuck Else TuckClosure xs, 0, -1, 0, L, W, tuck
    End Select
    If oy > 0 Then
        m = Min2(oy, Min2(L, H) / 2 - 5)
        If Has(fx, "OR") Then
            CutCircle (xs(2) + xs(3)) / 2, H / 2, Min2(L, H) / 2 - m
        Else
            CutRoundRect xs(2) + m, m * 1.5, xs(3) - m, H - m * 1.5, 4
        End If
    End If
    If il > 0 Then                               ' osma qism orqa panel ustida
        c = Min2(il / 3, 8)
        Poly xs(2), H, xs(3), H, xs(3), H + il - c, xs(3) - c, H + il, xs(2) + c, H + il, xs(2), H + il - c
        CutEuroSlot (xs(2) + xs(3)) / 2, H + il * 0.55, Min2(L * 0.4, 30)
    End If
    If Has(fx, "TS") Then                        ' yirtiladigan tasma (aylana bo'ylab)
        Perf 0, H * 0.8, xs(4), H * 0.8
        Perf 0, H * 0.8 - 6, xs(4), H * 0.8 - 6
    End If
    SetRoot (xs(2) + xs(3)) / 2, H / 2, True       ' 3D: old panel tik, qopqoq orqa panelda
    SetLid xs(0), H, xs(1), H
    Box3 0, 0, 0, L, H, W, False
    Lid3 L, H, W, 35, tuck
End Sub

' Qopqoq (yumaloq tilli) va quloqlar. Quloqning qopqoqqa yaqin tomoni tik, uzoq tomoni qiya
Private Sub TuckClosure(xs() As Double, ByVal b As Double, ByVal s As Double, ByVal lid As Long, _
                        ByVal L As Double, ByVal W As Double, ByVal tuck As Double)
    Dim i As Long, xa As Double, xb As Double, y1 As Double, y2 As Double
    Dim dh As Double, rr As Double, w_ As Double, c As Double, stp As Double, gap As Double
    dh = Min2(W * 0.62, L / 2 - 1)
    For i = 0 To 3
        xa = xs(i): xb = xs(i + 1)
        If i = lid Then
            y1 = b + s * W: y2 = b + s * (W + tuck)
            rr = Min2(tuck * 0.7, (xb - xa) / 4)
            Rect xa, b, xb, y1
            PtBegin
            Pt xa, y1
            If s > 0 Then
                ArcPt xa + rr, y2 - rr, rr, 180, 90
                ArcPt xb - rr, y2 - rr, rr, 90, 0
            Else
                ArcPt xa + rr, y2 + rr, rr, 180, 270
                ArcPt xb - rr, y2 + rr, rr, 270, 360
            End If
            Pt xb, y1
            PtPoly
        ElseIf i Mod 2 = 1 Then
            w_ = xb - xa: c = w_ * 0.2: stp = dh * 0.12: rr = Min2(dh * 0.15, w_ * 0.1): gap = 0.5
            PtBegin
            If (i + 3) Mod 4 = lid Then              ' qopqoq chap tomonda
                Pt xa, b: Pt xa + gap, b + s * (dh - rr)
                If s > 0 Then ArcPt xa + gap + rr, b + s * (dh - rr), rr, 180, 90 Else ArcPt xa + gap + rr, b + s * (dh - rr), rr, 180, 270
                Pt xb - c + rr, b + s * dh: Pt xb, b + s * stp: Pt xb, b
            Else
                Pt xa, b: Pt xa, b + s * stp: Pt xa + c - rr, b + s * dh
                If s > 0 Then ArcPt xb - gap - rr, b + s * (dh - rr), rr, 90, 0 Else ArcPt xb - gap - rr, b + s * (dh - rr), rr, 270, 360
                Pt xb - gap, b + s * (dh - rr): Pt xb, b
            End If
            PtPoly
        End If
    Next i
End Sub

' Qulf tag (1-2-3 snap lock): panellar L | W | L | W, pastga
Private Sub SnapBottom(xs() As Double, ByVal L As Double, ByVal W As Double)
    Dim dpt As Double, sl As Double, sl2 As Double, nd As Double, a As Double, b As Double
    dpt = 0.77 * W: sl = W / 2: nd = 0.26 * W
    a = xs(0): b = xs(1)
    Poly a, 0, a, 0 - dpt, a + L / 4, 0 - dpt, a + L / 4, nd - dpt, b - L / 4, nd - dpt, b - L / 4, 0 - dpt, b, 0 - dpt, b, 0
    a = xs(1): b = xs(2)
    Poly a, 0, a + sl, 0 - sl, a + sl, 0 - dpt, b, 0 - dpt, b, 0
    a = xs(2): b = xs(3)
    sl2 = Min2(sl, L / 3)                        ' kvadrat qutida ham pastki qismi tekis qolsin
    Poly a, 0, a + sl2, 0 - sl2, a + sl2, 0 - dpt, b - sl2, 0 - dpt, b - sl2, 0 - sl2, b, 0
    a = xs(3): b = xs(4)
    Poly a, 0, a, 0 - dpt, b - sl, 0 - dpt, b - sl, 0 - sl, b, 0
End Sub

' Yopishtiriladigan tag: L panellarda katta qanot, W panellarda quloq
Private Sub SealEnd(xs() As Double, ByVal L As Double, ByVal W As Double)
    Dim i As Long, a As Double, b As Double, f As Double, c As Double
    For i = 0 To 3
        a = xs(i): b = xs(i + 1)
        If i Mod 2 = 0 Then
            f = W * 0.75
            If i = 2 Then f = W * 0.45
            c = Min2(f * 0.3, (b - a) / 6)
            Poly a, 0, a + c, 0 - f, b - c, 0 - f, b, 0
        Else
            f = W * 0.45: c = f * 0.5
            Poly a, 0, a + 1, 0 - f + c, a + c, 0 - f, b - c, 0 - f, b - 1, 0 - f + c, b, 0
        End If
    Next i
End Sub

' ================================================================ ruchkali

' snap - 1-2-3 qulf tag, rnd - ruchka qismi yumaloq (arka)
Private Sub Gable(ByVal L As Double, ByVal W As Double, ByVal H As Double, ByVal g As Double, _
                  ByVal hh As Double, ByVal hw As Double, ByVal snap As Boolean, ByVal rnd As Boolean, ByVal oy As Double, _
                  ByVal fx As String)
    Dim xs() As Double, i As Long, xa As Double, xb As Double, y1 As Double, y2 As Double
    Dim c As Double, cx As Double, cy As Double, th As Double, xm As Double, m As Double
    hw = Min2(hw, L - 20)
    Body xs, H, g, L, W, L, W
    If oy > 0 Then                               ' old panelda oyna
        m = Min2(oy, L / 2 - 5)
        If Has(fx, "OR") Then
            CutCircle (xs(2) + xs(3)) / 2, H * 0.55, Min2(L, H) / 2 - m
        Else
            CutRoundRect xs(2) + m, H * 0.2, xs(3) - m, H - m, 4
        End If
    End If
    If snap Then SnapBottom xs, L, W Else SlotFlaps xs, 0, -1, W / 2
    y1 = H + W / 2: y2 = y1 + hh: c = Min2(hh / 3, 15)
    For i = 0 To 3
        xa = xs(i): xb = xs(i + 1)
        Rect xa, H, xb, y1
        If i Mod 2 = 0 Then
            SetFold xa, y1, xb, y1, -90                  ' ruchka qismi tik ko'tariladi
            cx = (xa + xb) / 2: cy = y1 + hh * 0.55: th = Min2(22, hh * 0.35)
            If rnd Then
                Dim ry As Double
                ry = Min2((xb - xa) / 2, hh * 0.8)
                PtBegin
                Pt xa, y1: Pt xb, y1: Pt xb, y2 - ry
                EllPt cx, y2 - ry, (xb - xa) / 2, ry, 0, 180, 16
                PtPoly
                cy = y2 - ry
            Else
                Poly xa, y1, xb, y1, xb, y2 - c, xb - c, y2, xa + c, y2, xa, y2 - c
            End If
            If hw > 0 Then CutStadium cx, cy, hw, th
        Else
            xm = (xa + xb) / 2
            Crease xa, H, xm, y1
            Crease xb, H, xm, y1
        End If
    Next i
    SetRoot (xs(2) + xs(3)) / 2, H / 2, True
    Box3 0, 0, 0, L, H, W, False
    Face 0, H, 0, 0, H + W / 2, W / 2, L, H + W / 2, W / 2, L, H, 0
    Face 0, H, W, L, H, W, L, H + W / 2, W / 2, 0, H + W / 2, W / 2
    Face 0, H, 0, 0, H, W, 0, H + W / 2, W / 2
    Face L, H, 0, L, H + W / 2, W / 2, L, H, W
    Face 0, H + W / 2, W / 2, L, H + W / 2, W / 2, L, H + W / 2 + hh, W / 2, 0, H + W / 2 + hh, W / 2
End Sub

' ================================================================ lotok

' Oddiy lotok: quloqlar yon devorlarga ulangan, old/orqa devorga yelimlanadi. ox: X bo'yicha siljish,
' rt - yon devorlarda ruchka teshigi eni
Private Sub Tray(ByVal ox As Double, ByVal L As Double, ByVal W As Double, ByVal H As Double, ByVal rt As Double)
    Dim x0 As Double, y0 As Double, sx As Double, sy As Double, eLen As Double, a As Long, b As Long
    eLen = Min2(H, L / 2 - 2)
    Rect ox, 0, ox + L, W
    Rect ox, 0 - H, ox + L, 0
    Rect ox, W, ox + L, W + H
    For a = 0 To 1
        If a = 0 Then x0 = ox: sx = -1 Else x0 = ox + L: sx = 1
        Rect x0, 0, x0 + sx * H, W
        For b = 0 To 1
            If b = 0 Then y0 = 0: sy = -1 Else y0 = W: sy = 1
            Poly x0, y0, x0 + sx * (H - 1), y0, x0 + sx * (H - 1), y0 + sy * (eLen - 2), _
                 x0 + sx * 2, y0 + sy * eLen, x0, y0 + sy * eLen
            Slit x0, y0, x0, y0 + sy * eLen
        Next b
        If rt > 0 And H >= 40 Then HandleSide x0 + sx * H * 0.6, W, rt, H
    Next a
End Sub

' Yon devordagi ruchka teshigi (devor bo'ylab vertikal)
Private Sub HandleSide(ByVal x As Double, ByVal W As Double, ByVal rt As Double, ByVal H As Double)
    Dim hw As Double, th As Double, r As Double
    hw = Min2(rt, W - 30): th = Min2(25, H * 0.3): r = th / 2
    PtBegin
    ArcPt x, W / 2 + hw / 2 - r, r, 0, 180
    ArcPt x, W / 2 - hw / 2 + r, r, 180, 360
    PtCut
End Sub

' Ikki qavat old/orqa devor (qulf tillari bilan), quloqlar yon devorlarda
Private Sub TrayDouble(ByVal L As Double, ByVal W As Double, ByVal H As Double, ByVal t As Double, ByVal rt As Double)
    Dim x0 As Double, y0 As Double, sx As Double, sy As Double, eLen As Double, a As Long, b As Long
    eLen = Min2(1.3 * H, L / 2 - 2)
    Rect 0, 0, L, W
    For a = 0 To 1
        If a = 0 Then x0 = 0: sx = -1 Else x0 = L: sx = 1
        Rect x0, 0, x0 + sx * H, W
        For b = 0 To 1
            If b = 0 Then y0 = 0: sy = -1 Else y0 = W: sy = 1
            Rect x0, y0, x0 + sx * (H - 1), y0 + sy * eLen
            Slit x0, y0, x0, y0 + sy * eLen
        Next b
        If rt > 0 And H >= 40 Then HandleSide x0 + sx * H * 0.6, W, rt, H
    Next a
    For b = 0 To 1
        If b = 0 Then y0 = 0: sy = -1 Else y0 = W: sy = 1
        Rect 0, y0, L, y0 + sy * H
        LockWall y0, sy, L, H, t, True
        BaseSlots y0, L, True
        SetFold 0, y0 + sy * H, L, y0 + sy * H, 180
    Next b
End Sub

' Muntazam ko'pburchak lotok. g > 0 - devorlar orasida yelim qanoti, pet > 0 - gulbargli devorlar (gul quti)
Private Sub PolyTray(ByVal n As Long, ByVal sd As Double, ByVal H As Double, ByVal g As Double, ByVal pet As Double)
    Dim k As Long, vx() As Double, vy() As Double
    If n < 3 Then n = 3
    RegPoly 0, 0, sd, 0, n
    ReDim vx(0 To n - 1): ReDim vy(0 To n - 1)
    For k = 0 To n - 1: vx(k) = ptX(k): vy(k) = ptY(k): Next k
    PolyWalls vx, vy, n, H, g, pet
    SetRoot sd / 2, sd / 2 * Tan(PI * (n - 2) / (2 * n)), False   ' taglik markazi
    PolyTray3D vx, vy, n, 0, H, True
End Sub

' Tortma quti: lotok + g'ilof. fx: W - lotok burchaklari bukiladigan, D - lotok ikki qavat,
' TN - g'ilof uchlarida barmoq o'yig'i, FH - g'ilofda barmoq teshiklari. oy > 0 - g'ilofda oyna
Private Sub Drawer(ByVal L As Double, ByVal W As Double, ByVal H As Double, ByVal g As Double, ByVal k As Double, _
                   ByVal oy As Double, ByVal fx As String)
    Dim ox As Double, ww As Double, hh As Double, i As Long, ls As Double, r As Double, cx As Double, m As Double
    Dim xs(0 To 4) As Double
    If Has(fx, "W") Then
        TrayWebAt 0, 0, L, W, H
    ElseIf Has(fx, "D") Then
        TrayDouble L, W, H, 2, 0
    Else
        Tray 0, L, W, H, 0
    End If
    ox = L + H + 30 + g
    ww = W + 2 * k: hh = H + k: ls = L + k
    xs(1) = ww: xs(2) = ww + hh: xs(3) = 2 * ww + hh: xs(4) = 2 * ww + 2 * hh
    Glue ox, ls, g
    For i = 0 To 3
        If i = 2 And Has(fx, "TN") Then          ' ustki panel: ikki uchida yarim doira o'yiq
            r = Min2(12, ww * 0.12): cx = ox + (xs(2) + xs(3)) / 2
            PtBegin
            Pt ox + xs(2), 0
            ArcPt cx, 0, r, 180, 0
            Pt ox + xs(3), 0: Pt ox + xs(3), ls
            ArcPt cx, ls, r, 0, -180
            Pt ox + xs(2), ls
            PtPoly
        Else
            Rect ox + xs(i), 0, ox + xs(i + 1), ls
        End If
    Next i
    cx = ox + (xs(2) + xs(3)) / 2
    If oy > 0 Then
        m = Min2(oy, Min2(ww, ls) / 2 - 5)
        CutRoundRect ox + xs(2) + m, m, ox + xs(3) - m, ls - m, 5
    End If
    If Has(fx, "FH") Then
        CutSlotV cx - 9, ls / 2, Min2(ls * 0.3, 40), 10
        CutSlotV cx + 9, ls / 2, Min2(ls * 0.3, 40), 10
    End If
    SetRoot L / 2, W / 2, False
    Box3 0, 0, 0, L, H, W, False
    Box3 L * 0.4, 0 - k, 0 - k, L + k, H + 2 * k, W + 2 * k, True, True, False
End Sub

' ================================================================ ko'p burchakli

' n burchakli prizma: n ta panel (tomoni sd, balandligi H). Qopqoq va tag - ko'pburchak, 0-panelga ulangan
Private Sub Prism(ByVal n As Long, ByVal sd As Double, ByVal H As Double, ByVal g As Double, ByVal tk As Double)
    Dim k As Long, s As Double
    If n < 3 Then n = 3
    Glue 0, H, g
    For k = 0 To n - 1: Rect k * sd, 0, (k + 1) * sd, H: Next k
    For k = 0 To 1
        If k = 0 Then s = 1 Else s = -1
        PrismEnd n, sd, H, tk, s
    Next k
    SetRoot sd / 2, H / 2, True                  ' 3D: panellar orasida tashqi burchak 360/n
    For k = 0 To n - 1: SetFold k * sd, 0, k * sd, H, 360 / n: Next k
    SetLid 0, H, sd, H, 0                        ' qopqoq quloqlar ustida
    ' 3D
    Dim vx() As Double, vz() As Double, j As Long
    RegPoly 0, 0, sd, 0, n
    ReDim vx(0 To n - 1): ReDim vz(0 To n - 1)
    For k = 0 To n - 1: vx(k) = ptX(k): vz(k) = ptY(k): Next k
    For k = 0 To n - 1
        j = (k + 1) Mod n
        Face vx(k), 0, vz(k), vx(j), 0, vz(j), vx(j), H, vz(j), vx(k), H, vz(k)
    Next k
    For k = 2 To n - 1
        Face vx(0), H, vz(0), vx(k), H, vz(k), vx(k - 1), H, vz(k - 1)
    Next k
End Sub

' Prizma uchi: s = 1 - yuqori (y = H), s = -1 - past (y = 0)
Private Sub PrismEnd(ByVal n As Long, ByVal sd As Double, ByVal H As Double, ByVal tk As Double, ByVal s As Double)
    Dim k As Long, b As Double, vx() As Double, vy() As Double, i As Long, j As Long
    Dim ex As Double, ey As Double, nx As Double, ny As Double, c As Double, dh As Double
    If s > 0 Then b = H: RegPoly 0, b, sd, b, n Else b = 0: RegPoly sd, b, 0, b, n
    ReDim vx(0 To n - 1): ReDim vy(0 To n - 1)
    For k = 0 To n - 1: vx(k) = ptX(k): vy(k) = ptY(k): Next k
    PolyArr vx, vy, n
    ' til qarama-qarshi qirrada
    i = n \ 2: j = (i + 1) Mod n
    ex = (vx(j) - vx(i)) / sd: ey = (vy(j) - vy(i)) / sd
    nx = ey: ny = -ex
    c = Min2(tk * 0.6, sd / 4)
    Poly vx(i), vy(i), vx(i) + nx * tk + ex * c, vy(i) + ny * tk + ey * c, _
         vx(j) + nx * tk - ex * c, vy(j) + ny * tk - ey * c, vx(j), vy(j)
    ' qolgan panellarda quloqlar
    dh = Min2(sd * 0.3, tk * 1.5)
    If n = 3 Then dh = sd * 0.2
    For k = 1 To n - 1
        Poly k * sd, b, k * sd + dh * 0.8, b + s * dh, (k + 1) * sd - dh * 0.8, b + s * dh, (k + 1) * sd, b
    Next k
End Sub

' n burchakli piramida: taglik tomoni sd, balandligi H, har uchburchak o'ng qirrasida yelim qanoti
' crv = True: yon tomonlar qavariq gulbarg, yelimsiz, uchida lenta teshigi
Private Sub Pyramid(ByVal n As Long, ByVal sd As Double, ByVal H As Double, ByVal g As Double, ByVal crv As Boolean, _
                    ByVal hole As Boolean, ByVal sc As Boolean)
    Dim k As Long, j As Long, vx() As Double, vy() As Double, rin As Double, sl As Double
    Dim ex As Double, ey As Double, nx As Double, ny As Double, axp As Double, ayp As Double
    Dim fx As Double, fy As Double, fl As Double, ox As Double, oy As Double, c As Double
    If n < 3 Then n = 3
    RegPoly 0, 0, sd, 0, n
    ReDim vx(0 To n - 1): ReDim vy(0 To n - 1)
    For k = 0 To n - 1: vx(k) = ptX(k): vy(k) = ptY(k): Next k
    PolyArr vx, vy, n
    rin = sd / (2 * Tan(PI / n))
    sl = Sqr(H * H + rin * rin)                  ' yon uchburchak balandligi
    For k = 0 To n - 1
        j = (k + 1) Mod n
        ex = (vx(j) - vx(k)) / sd: ey = (vy(j) - vy(k)) / sd
        nx = ey: ny = -ex
        axp = (vx(k) + vx(j)) / 2 + nx * sl: ayp = (vy(k) + vy(j)) / 2 + ny * sl
        SetFold vx(k), vy(k), vx(j), vy(j), 180 - Atn(H / rin) * 180 / PI
        If Not crv And g > 0 Then SetFold vx(j), vy(j), axp, ayp, FlapAng(n, rin, H)
        If hole And Not crv Then                 ' uchida halqa / lenta teshigi
            CutCircle axp + (((vx(k) + vx(j)) / 2) - axp) * 0.1, ayp + (((vy(k) + vy(j)) / 2) - ayp) * 0.1, 3
        End If
        If sc Then                               ' taroqli qirralar
            PtBegin
            Pt vx(k), vy(k): Pt vx(j), vy(j)
            ScallopPts vx(j), vy(j), axp, ayp
            ScallopPts axp, ayp, vx(k), vy(k)
            ptN = ptN - 1                        ' oxirgi nuqta birinchisi bilan bir xil
            PtPoly
        ElseIf crv Then
            CurvedTri vx(k), vy(k), vx(j), vy(j), axp, ayp, sl * 0.12
            CutCircle axp + (((vx(k) + vx(j)) / 2) - axp) * 0.12, ayp + (((vy(k) + vy(j)) / 2) - ayp) * 0.12, 2.5
        Else
            Poly vx(k), vy(k), vx(j), vy(j), axp, ayp
            If g > 0 Then
                ' yelim qanoti: vx(j) -> uchi qirrasi bo'ylab, tashqariga
                fx = axp - vx(j): fy = ayp - vy(j): fl = Sqr(fx * fx + fy * fy)
                fx = fx / fl: fy = fy / fl
                ox = 0 - fy: oy = fx                  ' qirradan tashqariga (uchburchak soat strelkasi bo'yicha)
                c = Min2(g, fl / 4)
                Poly vx(j), vy(j), vx(j) + ox * g + fx * c * 1.5, vy(j) + oy * g + fy * c * 1.5, _
                     axp + ox * g - fx * c * 2, ayp + oy * g - fy * c * 2, axp, ayp
            End If
        End If
    Next k
    ' 3D
    Dim cx As Double, cz As Double
    For k = 0 To n - 1: cx = cx + vx(k) / n: cz = cz + vy(k) / n: Next k
    SetRoot cx, cz, False
    For k = 0 To n - 1
        j = (k + 1) Mod n
        Face vx(k), 0, vy(k), vx(j), 0, vy(j), cx, H, cz
    Next k
End Sub

' ================================================================ sovg'a qutilari

' Gulbargli quti: kvadrat (sd x sd), tag 1-2-3 qulf. sty: A - yarim doira, H - yurak, P - o'tkir uchli
Private Sub Petal(ByVal sty As String, ByVal sd As Double, ByVal H As Double, ByVal g As Double, ByVal pet As Double)
    Dim xs() As Double, k As Long, xa As Double, xb As Double, xm As Double, r As Double, i As Long, t As Double
    Body xs, H, g, sd, sd, sd, sd
    SnapBottom xs, sd, sd
    For k = 0 To 3
        xa = xs(k): xb = xs(k + 1): xm = (xa + xb) / 2
        PtBegin
        Select Case sty
            Case "H"                             ' yurak: ikki bo'lak, ingichka bo'yin
                r = sd / 4
                Pt xa, H
                EllPt xm - r, H + pet - r, r, r, 225, 0, 16
                EllPt xm + r, H + pet - r, r, r, 180, -45, 16
                Pt xb, H
            Case "P"
                For i = 0 To 10
                    t = i / 10
                    Pt xa + sd / 2 * (1 - Sqr(1 - t * t)), H + pet * t
                Next i
                For i = 9 To 0 Step -1
                    t = i / 10
                    Pt xb - sd / 2 * (1 - Sqr(1 - t * t)), H + pet * t
                Next i
            Case "T"                             ' lentali: tik qism, keyin uchli
                Pt xa, H: Pt xa, H + pet * 0.45: Pt xm, H + pet: Pt xb, H + pet * 0.45: Pt xb, H
            Case Else
                EllPt xm, H, sd / 2, pet, 180, 0, 16
        End Select
        PtPoly
        If sty = "T" Then
            CutCircle xm - sd * 0.12, H + pet * 0.62, 2.5
            CutCircle xm + sd * 0.12, H + pet * 0.62, 2.5
        End If
    Next k
    ' 3D: kub va ichkariga egilgan gulbarglar
    SetRoot (xs(2) + xs(3)) / 2, H / 2, True       ' 3D: gulbarglar - qopqoq
    SetLid 0, H, xs(4), H
    Box3 0, 0, 0, sd, H, sd, False
    Dim a As Double, cs As Double, sn As Double
    a = 65 * PI / 180: cs = Cos(a): sn = Sin(a)
    For k = 0 To 3
        PetalFace k, sd, H, pet, cs, sn
    Next k
End Sub

' 3D da bitta gulbarg (soddalashtirilgan beshburchak)
Private Sub PetalFace(ByVal k As Long, ByVal sd As Double, ByVal H As Double, ByVal pet As Double, _
                      ByVal cs As Double, ByVal sn As Double)
    Dim x0 As Double, z0 As Double, dx As Double, dz As Double, ix As Double, iz As Double
    Select Case k
        Case 0: x0 = 0: z0 = sd: dx = 1: dz = 0: ix = 0: iz = -1
        Case 1: x0 = sd: z0 = sd: dx = 0: dz = -1: ix = -1: iz = 0
        Case 2: x0 = sd: z0 = 0: dx = -1: dz = 0: ix = 0: iz = 1
        Case Else: x0 = 0: z0 = 0: dx = 0: dz = 1: ix = 1: iz = 0
    End Select
    Dim v1 As Double, v2 As Double
    v1 = pet * 0.6: v2 = pet
    Face x0, H, z0, x0 + dx * sd, H, z0 + dz * sd, _
         x0 + dx * sd * 0.95 + ix * v1 * sn, H + v1 * cs, z0 + dz * sd * 0.95 + iz * v1 * sn, _
         x0 + dx * sd / 2 + ix * v2 * sn, H + v2 * cs, z0 + dz * sd / 2 + iz * v2 * sn, _
         x0 + dx * sd * 0.05 + ix * v1 * sn, H + v1 * cs, z0 + dz * sd * 0.05 + iz * v1 * sn
End Sub

Private Sub Pillow(ByVal L As Double, ByVal W As Double, ByVal g As Double, ByVal d As Double)
    Const N As Long = 16
    Dim i As Long, k As Long, xa As Double, xb As Double, t As Double
    Dim lx(0 To N) As Double, ltY(0 To N) As Double, htY(0 To N) As Double, lbY(0 To N) As Double, hbY(0 To N) As Double
    Dim ax() As Double, ay() As Double, m As Long
    d = Min2(d, W / 2)
    Glue 0, L, g
    For i = 0 To 1
        xa = i * W: xb = xa + W
        For k = 0 To N
            t = k / N
            lx(k) = xa + W * t
            ltY(k) = L + d * Sin(PI * t): htY(k) = L + 2 * d * Sin(PI * t)
            lbY(k) = -d * Sin(PI * t): hbY(k) = -2 * d * Sin(PI * t)
        Next k
        ' panel: past yoy -> yuqori yoy (teskari)
        ReDim ax(0 To 2 * N + 1): ReDim ay(0 To 2 * N + 1): m = 0
        For k = 0 To N: ax(m) = lx(k): ay(m) = lbY(k): m = m + 1: Next k
        For k = N To 0 Step -1: ax(m) = lx(k): ay(m) = ltY(k): m = m + 1: Next k
        PolyArr ax, ay, m
        ' qopqoqlar (linza shakli): ichki yoy bukish, tashqi yoy kesish
        m = 0
        For k = 0 To N: ax(m) = lx(k): ay(m) = ltY(k): m = m + 1: Next k
        For k = N - 1 To 1 Step -1: ax(m) = lx(k): ay(m) = htY(k): m = m + 1: Next k
        PolyArr ax, ay, m
        m = 0
        For k = 0 To N: ax(m) = lx(k): ay(m) = lbY(k): m = m + 1: Next k
        For k = N - 1 To 1 Step -1: ax(m) = lx(k): ay(m) = hbY(k): m = m + 1: Next k
        PolyArr ax, ay, m
    Next i
    ' 3D: x - uzunlik, y - qavariqlik, z - eni
    Dim th As Double, y0 As Double, y1 As Double, x0 As Double, x1 As Double
    th = d * 0.8
    For k = 0 To 11
        x0 = L * k / 12: x1 = L * (k + 1) / 12
        y0 = th * Sin(PI * k / 12): y1 = th * Sin(PI * (k + 1) / 12)
        Face x0, y0, 0, x0, y0, W, x1, y1, W, x1, y1, 0
        Face x0, 0 - y0, 0, x1, 0 - y1, 0, x1, 0 - y1, W, x0, 0 - y0, W
    Next k
End Sub

' Konfet qutisi: 4 panelli quvur (kesimi sd x sd, uzunligi L), uchlari E - buraladigan
Private Sub Candy(ByVal sd As Double, ByVal L As Double, ByVal e As Double, ByVal g As Double)
    Dim xs() As Double, k As Long, xa As Double, xb As Double, xm As Double, s As Double, b As Double, j As Long
    Body xs, L, g, sd, sd, sd, sd
    For j = 0 To 1
        If j = 0 Then b = L: s = 1 Else b = 0: s = -1
        For k = 0 To 3
            xa = xs(k): xb = xs(k + 1): xm = (xa + xb) / 2
            Poly xa, b, xb, b, xb, b + s * e, xa, b + s * e
            Crease xa, b, xm, b + s * e
            Crease xb, b, xm, b + s * e
        Next k
        For k = 1 To 3: Slit xs(k), b + s * e * 0.6, xs(k), b + s * e: Next k
    Next j
    ' 3D: quti va siqilgan uchlar
    Box3 0, 0, 0, L, sd, sd, False, True, False
    Face 0, sd, 0, 0, sd, sd, L, sd, sd, L, sd, 0
    Face 0, 0, 0, 0 - e, sd * 0.35, sd * 0.35, 0 - e, sd * 0.65, sd * 0.65, 0, sd, sd
    Face 0, sd, 0, 0 - e, sd * 0.65, sd * 0.35, 0 - e, sd * 0.35, sd * 0.65, 0, 0, sd
    Face L, 0, 0, L + e, sd * 0.35, sd * 0.35, L + e, sd * 0.65, sd * 0.65, L, sd, sd
    Face L, sd, 0, L + e, sd * 0.65, sd * 0.35, L + e, sd * 0.35, sd * 0.65, L, 0, sd
End Sub

' ================================================================ qo'shimcha yordamchilar

' Doira teshik (lenta uchun)
Private Sub CutCircle(ByVal cx As Double, ByVal cy As Double, ByVal r As Double)
    Dim n As Long
    n = Int(r) + 11                              ' katta doira silliqroq
    If n > 47 Then n = 47
    PtBegin
    EllPt cx, cy, r, r, 0, 360 - 360 / (n + 1), n
    PtCut
End Sub

' Uchburchak (a -> b asos, c uchi), yon tomonlari tashqariga bo'rtgan (bul - bo'rtiq)
Private Sub CurvedTri(ByVal ax As Double, ByVal ay As Double, ByVal bx As Double, ByVal by As Double, _
                      ByVal cx As Double, ByVal cy As Double, ByVal bul As Double)
    PtBegin
    Pt ax, ay
    BulgePts bx, by, cx, cy, bul
    BulgePts cx, cy, ax, ay, bul
    PtPoly
End Sub

' (x1,y1) dan (x2,y2) gacha chap tomonga bo'rtgan egri chiziq nuqtalari (birinchi nuqtasiz)
Private Sub BulgePts(ByVal x1 As Double, ByVal y1 As Double, ByVal x2 As Double, ByVal y2 As Double, ByVal bul As Double)
    Dim i As Long, t As Double, dx As Double, dy As Double, ln As Double, ox As Double, oy As Double
    dx = x2 - x1: dy = y2 - y1: ln = Sqr(dx * dx + dy * dy)
    ox = 0 - dy / ln: oy = dx / ln
    For i = 0 To 12
        t = i / 12
        Pt x1 + dx * t + ox * bul * Sin(PI * t), y1 + dy * t + oy * bul * Sin(PI * t)
    Next i
End Sub

' (x1,y1) dan (x2,y2) gacha chap tomonga yarim doira taroqlar (birinchi nuqtasiz, oxirgisi bilan)
Private Sub ScallopPts(ByVal x1 As Double, ByVal y1 As Double, ByVal x2 As Double, ByVal y2 As Double)
    Dim ln As Double, ux As Double, uy As Double, nb As Long, i As Long, a As Long, r As Double
    Dim cx As Double, cy As Double, f As Double
    ln = Sqr((x2 - x1) ^ 2 + (y2 - y1) ^ 2)
    ux = (x2 - x1) / ln: uy = (y2 - y1) / ln
    nb = Int(ln / 28 + 0.5)
    If nb < 3 Then nb = 3
    r = ln / nb / 2
    For i = 0 To nb - 1
        cx = x1 + ux * r * (2 * i + 1): cy = y1 + uy * r * (2 * i + 1)
        For a = 1 To 8
            f = PI * a / 8
            Pt cx - r * Cos(f) * ux - r * Sin(f) * uy, cy - r * Cos(f) * uy + r * Sin(f) * ux
        Next a
    Next i
End Sub

' Ko'pburchak taglik (soat strelkasiga teskari) va har qirrada devor. g > 0 - yelim qanoti,
' pet > 0 - devor ustida gulbarg va tirqish
Private Sub PolyWalls(vx() As Double, vy() As Double, ByVal n As Long, ByVal H As Double, ByVal g As Double, ByVal pet As Double)
    Dim k As Long, j As Long, ex As Double, ey As Double, nx As Double, ny As Double, sd As Double
    Dim x1 As Double, y1 As Double, x2 As Double, y2 As Double, c As Double
    Dim mx As Double, my As Double, a As Double, u As Double, v As Double, j2 As Long, fl As Double, ta As Double
    PolyArr vx, vy, n
    For k = 0 To n - 1
        j = (k + 1) Mod n
        x1 = vx(k): y1 = vy(k): x2 = vx(j): y2 = vy(j)
        sd = Sqr((x2 - x1) ^ 2 + (y2 - y1) ^ 2)
        ex = (x2 - x1) / sd: ey = (y2 - y1) / sd
        nx = ey: ny = 0 - ex                      ' tashqi normal
        Poly x1, y1, x2, y2, x2 + nx * H, y2 + ny * H, x1 + nx * H, y1 + ny * H
        If g > 0 Then
            ' keyingi devor bilan burilish burchagi (muntazam ko'pburchakda 360/n)
            j2 = (j + 1) Mod n
            u = vx(j2) - x2: v = vy(j2) - y2: fl = Sqr(u * u + v * v)
            ta = Ang2((ex * v - ey * u) / fl, (ex * u + ey * v) / fl)
            If ta >= PI / 2 - 0.01 Then c = Min2(g * 0.3, H / 4) Else c = Min2(g / Tan(ta) * 1.1 + 1, H / 2)
            Poly x2, y2, x2 + ex * g + nx * c, y2 + ey * g + ny * c, _
                 x2 + ex * g + nx * (H - c), y2 + ey * g + ny * (H - c), x2 + nx * H, y2 + ny * H
            SetFold x2, y2, x2 + nx * H, y2 + ny * H, ta * 180 / PI      ' qo'shni devor ichiga
        End If
        If pet > 0 Then
            PtBegin
            For a = 0 To 12
                u = sd / 2 - sd / 2 * Cos(PI * a / 12): v = H + pet * Sin(PI * a / 12)
                Pt x1 + ex * u + nx * v, y1 + ey * u + ny * v
            Next a
            PtPoly
            mx = (x1 + x2) / 2: my = (y1 + y2) / 2
            CutPoly mx + nx * (H + pet * 0.55), my + ny * (H + pet * 0.55), mx + nx * (H + pet * 0.95), my + ny * (H + pet * 0.95)
        End If
    Next k
End Sub

' 3D: ko'pburchak lotok (y0 balandlikda)
Private Sub PolyTray3D(vx() As Double, vy() As Double, ByVal n As Long, ByVal y0 As Double, ByVal H As Double, ByVal bottomOn As Boolean)
    Dim k As Long, j As Long
    For k = 0 To n - 1
        j = (k + 1) Mod n
        Face vx(k), y0, vy(k), vx(j), y0, vy(j), vx(j), y0 + H, vy(j), vx(k), y0 + H, vy(k)
    Next k
    If bottomOn Then
        For k = 2 To n - 1
            Face vx(0), y0, vy(0), vx(k - 1), y0, vy(k - 1), vx(k), y0, vy(k)
        Next k
    End If
End Sub

' ================================================================ tort va lanch-boks

' To'rt bargli tort qutisi: kvadrat taglik, old/orqa uzun bargli devorlar (uchida til), yon tomonlar - linza.
' rt > 0 - tillarda ruchka teshigi
Private Sub Cross(ByVal sd As Double, ByVal H As Double, ByVal pet As Double, ByVal rt As Double)
    Dim la As Double, ls As Double, bw As Double, bh As Double, nw As Double, i As Long, t As Double
    Dim x As Double, w As Double, rr As Double, k As Long, sg As Double, y0 As Double
    la = H + sd * 0.55: ls = H + sd * 0.5
    bw = sd * 0.2: bh = sd * 0.35: nw = sd * 0.45
    Rect 0, 0, sd, sd
    For k = 0 To 1                               ' old va orqa barglar (yuqori va past)
        If k = 0 Then sg = 1: y0 = sd Else sg = -1: y0 = 0
        rr = Min2(nw * 0.35, pet / 2)
        PtBegin
        For i = 0 To 14                          ' chap qirra
            t = i / 14
            x = 0 - ArmBulge(t, bw): If t > 0.7 Then x = (sd - nw) / 2 * (t - 0.7) / 0.3
            Pt x, y0 + sg * la * t
        Next i
        Pt (sd - nw) / 2, y0 + sg * (la + pet - rr)
        If sg > 0 Then
            ArcPt (sd - nw) / 2 + rr, y0 + la + pet - rr, rr, 180, 90
            ArcPt (sd + nw) / 2 - rr, y0 + la + pet - rr, rr, 90, 0
        Else
            ArcPt (sd - nw) / 2 + rr, y0 - la - pet + rr, rr, 180, 270
            ArcPt (sd + nw) / 2 - rr, y0 - la - pet + rr, rr, 270, 360
        End If
        For i = 14 To 0 Step -1                  ' o'ng qirra
            t = i / 14
            x = sd + ArmBulge(t, bw): If t > 0.7 Then x = sd - (sd - nw) / 2 * (t - 0.7) / 0.3
            Pt x, y0 + sg * la * t
        Next i
        PtPoly
        If rt > 0 Then CutStadium sd / 2, y0 + sg * (la + pet * 0.5), Min2(rt, nw - 8), Min2(12, pet * 0.4)
    Next k
    For k = 0 To 1                               ' yon linzalar
        If k = 0 Then sg = 1: y0 = sd Else sg = -1: y0 = 0
        PtBegin
        For i = 0 To 16
            t = i / 16
            w = sd / 2 * (1 - t) + bh * Sin(PI * t)
            Pt y0 + sg * ls * t, sd / 2 - w
        Next i
        For i = 15 To 1 Step -1
            t = i / 16
            w = sd / 2 * (1 - t) + bh * Sin(PI * t)
            Pt y0 + sg * ls * t, sd / 2 + w
        Next i
        PtPoly
        CutPoly y0 + sg * ls * 0.62, sd / 2, y0 + sg * ls * 0.9, sd / 2
    Next k
    Box3 0, 0, 0, sd, H, sd
End Sub

' Bargning yon bo'rtig'i: tagida bo'rtmaydi (yon linzalar bilan to'qnashmasin), o'rtada eng keng
Private Function ArmBulge(ByVal t As Double, ByVal bw As Double) As Double
    If t > 0.15 And t <= 0.7 Then ArmBulge = bw * Sin(PI * (t - 0.15) / 0.55)
End Function

' Yumaloq uchli tort qutisi (taglik L x W, yon qanotlar yumaloq, ustki qopqoq uzun, boshi yarim doira)
Private Sub Cake(ByVal L As Double, ByVal W As Double, ByVal H As Double)
    Dim e As Double, r As Double, fl As Double, i As Long, t As Double, k As Long, x0 As Double, sx As Double
    Dim yt As Double, a As Double, cx As Double, ex As Double
    Rect 0, 0, L, W
    e = H + W * 0.4: r = W / 2: fl = H * 0.2
    For k = 0 To 1                               ' yon qanotlar
        If k = 0 Then x0 = 0: sx = -1 Else x0 = L: sx = 1
        PtBegin
        Pt x0, 0: Pt x0 + sx * (e - r), 0
        If sx < 0 Then ArcPt x0 - (e - r), r, r, 270, 90 Else ArcPt x0 + (e - r), r, r, 270, 450
        Pt x0 + sx * (e - r), W: Pt x0, W
        PtPoly
        cx = x0 + sx * e * 0.62: ex = e * 0.16    ' egilish uchun yoysimon bukish
        For i = 0 To 7
            Crease cx + sx * ex * Sin(PI * i / 8), W / 2 - r * Cos(PI * i / 8), _
                   cx + sx * ex * Sin(PI * (i + 1) / 8), W / 2 - r * Cos(PI * (i + 1) / 8)
        Next i
    Next k
    ' ustki qanot: devor (H) + qopqoq (W) + yarim doira boshi
    yt = W + H + W * 0.85
    PtBegin
    For i = 0 To 8: t = i / 8: Pt 0 - fl * Sin(PI * t), W + H * t: Next i
    Pt 0, yt
    EllPt L / 2, yt, L / 2, L * 0.3, 180, 0, 16
    Pt L, W + H
    For i = 7 To 1 Step -1: t = i / 8: Pt L + fl * Sin(PI * t), W + H * t: Next i
    Pt L, W
    PtPoly
    Crease 0, W + H, L, W + H
    CutStadium L / 2, yt + L * 0.12, L * 0.18, 5
    ' pastki qanot: devor (H) + til, tirqish bilan
    PtBegin
    Pt 0, 0
    For i = 1 To 8: t = i / 8: Pt 0 - fl * Sin(PI * t), 0 - H * t: Next i
    Pt 0, 0 - H - W * 0.45: Pt L, 0 - H - W * 0.45
    For i = 8 To 1 Step -1: t = i / 8: Pt L + fl * Sin(PI * t), 0 - H * t: Next i
    Pt L, 0
    PtPoly
    Crease 0, 0 - H, L, 0 - H
    CutPoly L * 0.41, 0 - H - W * 0.25, L * 0.59, 0 - H - W * 0.25
    Box3 0, 0, 0, L, H, W
End Sub

' Burchaklari bukiladigan lotok (web corner): burchak kvadrati diagonal bo'ylab bukiladi. (ox, oy) - taglik burchagi
Private Sub TrayWebAt(ByVal ox As Double, ByVal oy As Double, ByVal L As Double, ByVal W As Double, ByVal H As Double, _
                      Optional ByVal tn As Double = 0)
    Dim a As Long, b As Long, x0 As Double, y0 As Double, sx As Double, sy As Double
    Rect ox, oy, ox + L, oy + W
    NotchRect ox, oy, ox + L, oy - H, tn
    Rect ox, oy + W, ox + L, oy + W + H
    Rect ox - H, oy, ox, oy + W
    Rect ox + L, oy, ox + L + H, oy + W
    For a = 0 To 1
        If a = 0 Then x0 = ox: sx = -1 Else x0 = ox + L: sx = 1
        For b = 0 To 1
            If b = 0 Then y0 = oy: sy = -1 Else y0 = oy + W: sy = 1
            Poly x0, y0, x0 + sx * H, y0, x0 + sx * H, y0 + sy * H
            Poly x0, y0, x0 + sx * H, y0 + sy * H, x0, y0 + sy * H
            SetFold x0, y0, x0 + sx * H, y0 + sy * H, 180      ' burchak diagonal bo'ylab ikki buklanadi
        Next b
    Next a
End Sub

Private Sub TrayWeb(ByVal ox As Double, ByVal L As Double, ByVal W As Double, ByVal H As Double)
    TrayWebAt ox, 0, L, W, H
End Sub

' Lanch-boks: taglik va qopqoq (ikkalasi bukiladigan burchakli), orqa devor orqali ulangan, old tilli
Private Sub Lunch(ByVal L As Double, ByVal W As Double, ByVal H As Double)
    Dim tw As Double, th As Double, yt As Double, r As Double, cx As Double
    TrayWebAt 0, 0, L, W, H
    TrayWebAt 0, W + 2 * H, L, W, H
    Slit 0 - H, W + H, 0, W + H
    Slit L, W + H, L + H, W + H
    tw = Min2(L * 0.25, 50): th = Min2(H * 0.5, 25): yt = 2 * W + 3 * H: cx = L / 2: r = Min2(th * 0.5, tw / 3)
    PtBegin
    Pt cx - tw / 2, yt
    ArcPt cx - tw / 2 + r, yt + th - r, r, 180, 90
    ArcPt cx + tw / 2 - r, yt + th - r, r, 90, 0
    Pt cx + tw / 2, yt
    PtPoly
    CutPoly cx - tw / 2 - 1, 0 - H + 4, cx + tw / 2 + 1, 0 - H + 4
    SetRoot L / 2, W / 2, False                  ' 3D: qopqoq orqa devor ustida, yopiqda tik davom etadi
    SetFold 0, W + H, L, W + H, 0
    SetLid 0, W + H, L, W + H
    Box3 0, 0, 0, L, H, W, False
    Box3 0, H * 1.4, 0, L, H, W, True, False
End Sub

' Ko'pburchak lanch-boks: taglik va qopqoq lotoklari, bitta devor orqali ulangan
Private Sub PolyLunch(ByVal n As Long, ByVal sd As Double, ByVal H As Double, ByVal g As Double)
    Dim bx() As Double, by() As Double, lx() As Double, ly() As Double, k As Long
    If n < 3 Then n = 3
    RegPoly 0, 0, sd, 0, n
    ReDim bx(0 To n - 1): ReDim by(0 To n - 1): ReDim lx(0 To n - 1): ReDim ly(0 To n - 1)
    For k = 0 To n - 1: bx(k) = ptX(k): by(k) = ptY(k): Next k
    For k = 0 To n - 1                           ' qopqoq: devor tashqi chizig'iga nisbatan ko'zgu
        lx(k) = bx(n - 1 - k): ly(k) = 0 - 2 * H - by(n - 1 - k)
    Next k
    PolyWalls bx, by, n, H, g, 0
    PolyWalls lx, ly, n, H, g, 0
    SetRoot sd / 2, sd / 2 * Tan(PI * (n - 2) / (2 * n)), False
    SetFold 0, 0 - H, sd, 0 - H, 0                ' qopqoq taglik devori ustida davom etadi
    SetLid 0, 0 - H, sd, 0 - H
    PolyTray3D bx, by, n, 0, H, True
    For k = 0 To n - 1: ly(k) = by(k): Next k
    PolyTray3D bx, ly, n, H * 1.5, H, False
End Sub

' Bo'lakli vstavka: (ny-1) ta uzun va (nx-1) ta qisqa tasma, o'zaro kesishadigan tirqishlar bilan
Private Sub Insert(ByVal L As Double, ByVal W As Double, ByVal H As Double, ByVal nx As Long, ByVal ny As Long, ByVal t As Double)
    Dim j As Long, k As Long, y0 As Double, x As Double
    If nx < 1 Then nx = 1
    If ny < 1 Then ny = 1
    t = Max2(t, 1)
    y0 = 0
    For j = 1 To ny - 1                          ' uzun tasmalar: tirqish yuqoridan
        PtBegin
        Pt 0, y0: Pt L, y0: Pt L, y0 + H
        For k = nx - 1 To 1 Step -1
            x = L * k / nx
            Pt x + t / 2, y0 + H: Pt x + t / 2, y0 + H / 2: Pt x - t / 2, y0 + H / 2: Pt x - t / 2, y0 + H
        Next k
        Pt 0, y0 + H
        PtPoly
        y0 = y0 + H + 10
    Next j
    For j = 1 To nx - 1                          ' qisqa tasmalar: tirqish pastdan
        PtBegin
        Pt 0, y0
        For k = 1 To ny - 1
            x = W * k / ny
            Pt x - t / 2, y0: Pt x - t / 2, y0 + H / 2: Pt x + t / 2, y0 + H / 2: Pt x + t / 2, y0
        Next k
        Pt W, y0: Pt W, y0 + H: Pt 0, y0 + H
        PtPoly
        y0 = y0 + H + 10
    Next j
    For j = 1 To ny - 1: Face 0, 0, W * j / ny, L, 0, W * j / ny, L, H, W * j / ny, 0, H, W * j / ny: Next j
    For k = 1 To nx - 1: Face L * k / nx, 0, 0, L * k / nx, 0, W, L * k / nx, H, W, L * k / nx, H, 0: Next k
    Box3 0, 0, 0, L, H * 0.3, W, False, True, True
End Sub

' Uycha / fonar: kvadrat (sd) 4 devor, ustida to'rt qiyali tom (balandligi R), tag 1-2-3 qulf, oy > 0 - derazalar
Private Sub House(ByVal sd As Double, ByVal H As Double, ByVal R As Double, ByVal g As Double, ByVal oy As Double)
    Dim xs() As Double, k As Long, xa As Double, xb As Double, xm As Double, sl As Double
    Dim fx As Double, fy As Double, fl As Double, ox As Double, oyy As Double, c As Double
    Dim pw As Double, ph As Double, gp As Double, y1 As Double, i As Long, j As Long
    Body xs, H, g, sd, sd, sd, sd
    SnapBottom xs, sd, sd
    sl = Sqr(R * R + sd * sd / 4)
    For k = 0 To 3
        xa = xs(k): xb = xs(k + 1): xm = (xa + xb) / 2
        Poly xa, H, xb, H, xm, H + sl
        SetFold xa, H, xb, H, Atn(sd / (2 * R)) * 180 / PI       ' tom ichkariga egiladi
        SetFold xb, H, xm, H + sl, FlapAng(4, sd / 2, R)
        fx = xm - xb: fy = sl: fl = Sqr(fx * fx + fy * fy)
        fx = fx / fl: fy = fy / fl
        ox = fy: oyy = 0 - fx                     ' o'ng qirradan tashqariga
        c = Min2(g, fl / 4)
        Poly xb, H, xb + ox * g + fx * c * 1.5, H + oyy * g + fy * c * 1.5, _
             xm + ox * g - fx * c * 2, H + sl + oyy * g - fy * c * 2, xm, H + sl
        If oy > 0 Then                            ' 2 x 2 deraza
            gp = oy * 0.6
            pw = (sd - 2 * oy - gp) / 2: ph = (H * 0.7 - 2 * oy - gp) / 2: y1 = H * 0.22
            If pw > 3 And ph > 3 Then
                For i = 0 To 1
                    For j = 0 To 1
                        CutRoundRect xa + oy + i * (pw + gp), y1 + oy + j * (ph + gp), _
                                     xa + oy + i * (pw + gp) + pw, y1 + oy + j * (ph + gp) + ph, 2
                    Next j
                Next i
            End If
        End If
    Next k
    SetRoot (xs(2) + xs(3)) / 2, H / 2, True
    Box3 0, 0, 0, sd, H, sd, False
    Face 0, H, sd, sd, H, sd, sd / 2, H + R, sd / 2
    Face sd, H, sd, sd, H, 0, sd / 2, H + R, sd / 2
    Face sd, H, 0, 0, H, 0, sd / 2, H + R, sd / 2
    Face 0, H, 0, 0, H, sd, sd / 2, H + R, sd / 2
End Sub

' Ikki piramida (olmos): n burchakli ekvator (tomoni sd), ustki va pastki uchi H balandlikda.
' Yoyilma: ekvator bo'ylab n ta yuqori va n ta pastki uchburchak, har birining o'ng qirrasida yelim qanoti
Private Sub Bipyramid(ByVal n As Long, ByVal sd As Double, ByVal H As Double, ByVal g As Double)
    Dim k As Long, j As Long, sl As Double, xa As Double, xb As Double, xm As Double, s As Double
    Dim fx As Double, fy As Double, fl As Double, ox As Double, oy As Double, c As Double
    If n < 3 Then n = 3
    sl = Sqr(H * H + (sd / (2 * Tan(PI / n))) ^ 2)
    For j = 0 To 1
        If j = 0 Then s = 1 Else s = -1
        For k = 0 To n - 1
            xa = k * sd: xb = xa + sd: xm = xa + sd / 2
            Poly xa, 0, xb, 0, xm, s * sl
            fx = xm - xb: fy = s * sl: fl = Sqr(fx * fx + fy * fy)
            fx = fx / fl: fy = fy / fl
            ox = s * fy: oy = 0 - s * fx                ' o'ng qirradan tashqariga
            c = Min2(g, fl / 4)
            Poly xb, 0, xb + ox * g + fx * c * 1.5, oy * g + fy * c * 1.5, _
                 xm + ox * g - fx * c * 2, s * sl + oy * g - fy * c * 2, xm, s * sl
        Next k
    Next j
    ' 3D
    Dim vx() As Double, vz() As Double, cx As Double, cz As Double
    RegPoly 0, 0, sd, 0, n
    ReDim vx(0 To n - 1): ReDim vz(0 To n - 1)
    For k = 0 To n - 1: vx(k) = ptX(k): vz(k) = ptY(k): cx = cx + vx(k) / n: cz = cz + vz(k) / n: Next k
    For k = 0 To n - 1
        j = (k + 1) Mod n
        Face vx(k), 0, vz(k), vx(j), 0, vz(j), cx, H, cz
        Face vx(k), 0, vz(k), cx, 0 - H, cz, vx(j), 0, vz(j)
    Next k
End Sub

' Silindr quti: o'ralgan devor (aylana uzunligi), ikki uchida 8 tadan ichkariga bukiladigan qanot,
' alohida ikkita doira qopqoq
Private Sub Cylinder(ByVal dia As Double, ByVal H As Double, ByVal g As Double)
    Dim cl As Double, k As Long, nf As Long, fw As Double, fh As Double, x As Double, j As Long, b As Double, s As Double
    cl = PI * dia: nf = 8: fw = cl / nf: fh = Min2(dia * 0.15, 15)
    Glue 0, H, g
    Rect 0, 0, cl, H
    For j = 0 To 1
        If j = 0 Then b = H: s = 1 Else b = 0: s = -1
        For k = 0 To nf - 1
            x = k * fw
            Poly x, b, x + fw, b, x + fw - fh * 0.4, b + s * fh, x + fh * 0.4, b + s * fh
        Next k
    Next j
    For j = 0 To 1                               ' doira qopqoqlar yoyilma ostida
        PtBegin
        EllPt dia / 2 + j * (dia + 15), 0 - fh - 15 - dia / 2, dia / 2, dia / 2, 0, 350, 35
        PtPoly
    Next j
    ' 3D
    Dim a0 As Double, a1 As Double
    For k = 0 To 15
        a0 = 2 * PI * k / 16: a1 = 2 * PI * (k + 1) / 16
        Face dia / 2 * Cos(a0), 0, dia / 2 * Sin(a0), dia / 2 * Cos(a1), 0, dia / 2 * Sin(a1), _
             dia / 2 * Cos(a1), H, dia / 2 * Sin(a1), dia / 2 * Cos(a0), H, dia / 2 * Sin(a0)
    Next k
End Sub

' ================================================================ 1-bosqich: yangi quruvchilar

' Bukiladigan burchakli lotok + orqa devorga ulangan qopqoq (old tili, yon quloqlari).
' fx: TN - old devorda barmoq o'yig'i, LR - qopqoq quloqlari yumaloq. oy > 0 - qopqoqda oyna
Private Sub WebLid(ByVal L As Double, ByVal W As Double, ByVal H As Double, ByVal oy As Double, ByVal fx As String)
    Dim dd As Double, k As Double, y1 As Double, y2 As Double, ht As Double, rr As Double, m As Double
    TrayWebAt 0, 0, L, W, H, TnR(fx, L)
    y1 = W + H: y2 = 2 * W + H
    Rect 0, y1, L, y2                            ' qopqoq
    dd = H - 2: k = dd * 0.35
    If Has(fx, "LR") Then
        SideFlapR 0, y1 + 1, y2 - 1, -1, dd
        SideFlapR L, y1 + 1, y2 - 1, 1, dd
    Else
        Poly 0, y1 + 1, 0 - dd, y1 + 1 + k, 0 - dd, y2 - 1 - k, 0, y2 - 1
        Poly L, y1 + 1, L + dd, y1 + 1 + k, L + dd, y2 - 1 - k, L, y2 - 1
    End If
    ht = H * 0.85: rr = Min2(ht * 0.6, L / 4)     ' old til
    PtBegin
    Pt 0, y2
    ArcPt rr, y2 + ht - rr, rr, 180, 90
    ArcPt L - rr, y2 + ht - rr, rr, 90, 0
    Pt L, y2
    PtPoly
    If oy > 0 Then
        m = Min2(oy, Min2(L, W) / 2 - 5)
        CutRoundRect m, y1 + m, L - m, y2 - m, 5
    End If
    SetRoot L / 2, W / 2, False
    SetLid 0, y1, L, y1
End Sub

' Qiya devorli lotok: devor tepasi har tomonga f ga kengayadi (H - devorning qiya balandligi),
' yon devorlarda yelim quloqlari (eni g)
Private Sub Taper(ByVal L As Double, ByVal W As Double, ByVal H As Double, ByVal f As Double, ByVal g As Double)
    Dim al As Double, fa As Double, c As Double
    f = Min2(f, H * 0.4)
    g = Min2(g, H * 0.4)
    Rect 0, 0, L, W
    Poly 0, 0, L, 0, L + f, 0 - H, 0 - f, 0 - H
    Poly 0, W, 0 - f, W + H, L + f, W + H, L, W
    Poly 0, 0, 0, W, 0 - H, W + f, 0 - H, 0 - f
    Poly L, 0, L + H, 0 - f, L + H, W + f, L, W
    c = g * 1.2
    EdgeFlap 0 - H, 0 - f, 0, 0, g, c
    EdgeFlap 0, W, 0 - H, W + f, g, c
    EdgeFlap L, 0, L + H, 0 - f, g, c
    EdgeFlap L + H, W + f, L, W, g, c
    ' 3D: devor qiyaligi va quloqning qo'shni devorga burilishi
    al = Acs(f / H)
    fa = Acs(Cos(al) * Cos(al)) * 180 / PI
    al = al * 180 / PI
    SetRoot L / 2, W / 2, False
    SetFold 0, 0, L, 0, al: SetFold 0, W, L, W, al
    SetFold 0, 0, 0, W, al: SetFold L, 0, L, W, al
    SetFold 0 - H, 0 - f, 0, 0, fa: SetFold 0, W, 0 - H, W + f, fa
    SetFold L, 0, L + H, 0 - f, fa: SetFold L + H, W + f, L, W, fa
End Sub

' Burchaklari kesilgan lotok: L x W, burchak kesigi c (c = W/2 - olti burchak), devor H, yelim qanoti g
Private Sub ChTray(ByVal L As Double, ByVal W As Double, ByVal H As Double, ByVal c As Double, ByVal g As Double)
    Dim vx() As Double, vy() As Double, n As Long, k As Long, j As Long
    c = Min2(c, Min2(L, W) / 2)
    PtBegin
    Pt c, 0: Pt L - c, 0: Pt L, c: Pt L, W - c: Pt L - c, W: Pt c, W: Pt 0, W - c: Pt 0, c
    ReDim vx(0 To 7): ReDim vy(0 To 7)
    For k = 0 To 7                               ' ustma-ust tushgan uchlarni tashlab yuborish
        j = (k + 7) Mod 8
        If Abs(ptX(k) - ptX(j)) + Abs(ptY(k) - ptY(j)) > 0.01 Then vx(n) = ptX(k): vy(n) = ptY(k): n = n + 1
    Next k
    PolyWalls vx, vy, n, H, g, 0
    SetRoot L / 2, W / 2, False
    PolyTray3D vx, vy, n, 0, H, True
End Sub

' Ruchkali lotok: old/orqa devorlar ruchka qismi bilan baland (ht), quloqlar shu devorlarda.
' fx: RS - yon devorlar yarim doira
Private Sub HTray(ByVal L As Double, ByVal W As Double, ByVal H As Double, ByVal ht As Double, ByVal fx As String)
    Dim a As Long, b As Long, x0 As Double, sx As Double, y0 As Double, sy As Double, eLen As Double, c As Double
    eLen = Min2(H * 0.9, W / 2 - 2)
    Rect 0, 0, L, W
    For b = 0 To 1
        If b = 0 Then y0 = 0: sy = -1 Else y0 = W: sy = 1
        Rect 0, y0, L, y0 + sy * H
        If ht > 0 Then                           ' ruchka qismi: ichkariga egilib tepada uchrashadi
            c = Min2(ht / 2, L / 6)
            Poly 0, y0 + sy * H, L, y0 + sy * H, L, y0 + sy * (H + ht - c), L - c, y0 + sy * (H + ht), _
                 c, y0 + sy * (H + ht), 0, y0 + sy * (H + ht - c)
            CutStadium L / 2, y0 + sy * (H + ht * 0.55), Min2(L * 0.45, 100), Min2(25, ht * 0.35)
            SetFold 0, y0 + sy * H, L, y0 + sy * H, Asn(Min2(1, W / 2 / ht)) * 180 / PI
        End If
        For a = 0 To 1
            If a = 0 Then x0 = 0: sx = -1 Else x0 = L: sx = 1
            Ear x0, y0, sx, sy, eLen, H - 1, ""
        Next a
    Next b
    For a = 0 To 1
        If a = 0 Then x0 = 0: sx = -1 Else x0 = L: sx = 1
        PtBegin
        If Has(fx, "RS") Then
            If sx < 0 Then EllPt x0, W / 2, H, W / 2, 270, 90, 16 Else EllPt x0, W / 2, H, W / 2, -90, 90, 16
        Else
            Pt x0, 0: Pt x0 + sx * H, 0: Pt x0 + sx * H, W: Pt x0, W
        End If
        PtPoly
    Next a
    SetRoot L / 2, W / 2, False
End Sub

' Doira quti: old va orqa yuzasi doira (diametri dia), yon tasma (eni t) tishli qanotlar bilan, tepada osma halqa
Private Sub Disc(ByVal dia As Double, ByVal t As Double, ByVal g As Double)
    Dim r As Double, a As Double, h As Double, ang As Double, bl As Double, x As Double, k As Long, n As Long
    Dim tw As Double, s As Double, j As Long, b As Double, i As Long, ro As Double, yc As Double
    r = dia / 2: a = dia * 0.25: h = Sqr(r * r - a * a)
    ang = Atn(h / a) * 180 / PI
    Rect 0 - a, 0, a, t                          ' taglik
    ro = Min2(dia * 0.12, 15): yc = t + h + r + ro * 0.9
    PtBegin                                      ' old doira + osma halqa
    EllPt 0, t + h, r, r, 0 - ang, 80, 24
    EllPt 0, yc, ro, ro, -30, 210, 16
    EllPt 0, t + h, r, r, 100, 180 + ang, 24
    PtPoly
    CutCircle 0, yc, ro * 0.5
    PtBegin                                      ' orqa doira
    EllPt 0, 0 - h, r, r, ang, -180 - ang, 32
    PtPoly
    bl = (PI * dia - 2 * a) / 2                  ' yon tasmalar
    Rect 0 - a - bl, 0, 0 - a, t
    Rect a, 0, a + bl, t
    Glue 0 - a - bl, t, g
    n = Int(bl / 14)
    If n < 1 Then n = 1
    tw = bl / n
    For j = 0 To 1
        If j = 0 Then x = 0 - a - bl Else x = a
        For i = 0 To 1
            If i = 0 Then b = t: s = 1 Else b = 0: s = -1
            For k = 0 To n - 1
                Poly x + k * tw + 1, b, x + (k + 1) * tw - 1, b, x + (k + 1) * tw - 3, b + s * 6, x + k * tw + 3, b + s * 6
            Next k
        Next i
    Next j
    ' 3D: tik turgan doira
    Dim a0 As Double, a1 As Double
    For k = 0 To 7
        a0 = 2 * PI * k / 8: a1 = 2 * PI * (k + 1) / 8
        Face r * Cos(a0), r + r * Sin(a0), 0, r * Cos(a1), r + r * Sin(a1), 0, _
             r * Cos(a1), r + r * Sin(a1), t, r * Cos(a0), r + r * Sin(a0), t
    Next k
    Dim ex(0 To 7) As Double, ey(0 To 7) As Double
    For k = 0 To 7
        ex(k) = r * Cos(2 * PI * k / 8): ey(k) = r + r * Sin(2 * PI * k / 8)
    Next k
    For j = 0 To 1
        If j = 0 Then b = 0 Else b = t
        Face ex(0), ey(0), b, ex(1), ey(1), b, ex(2), ey(2), b, ex(3), ey(3), b, _
             ex(4), ey(4), b, ex(5), ey(5), b, ex(6), ey(6), b, ex(7), ey(7), b
    Next j
End Sub

' Kosmetika displeyi: orqa devor (tepasi yumaloq, balandligi hb), teshikli supa L x W x H (nx x ny teshik)
Private Sub Display(ByVal L As Double, ByVal W As Double, ByVal H As Double, ByVal hb As Double, _
                    ByVal nx As Long, ByVal ny As Long)
    Dim i As Long, j As Long, rr As Double, ry As Double
    If nx < 1 Then nx = 1
    If ny < 1 Then ny = 1
    ry = Min2(L / 2, hb * 0.35)
    PtBegin                                      ' orqa devor
    Pt 0, 0: Pt L, 0
    EllPt L / 2, hb - ry, L / 2, ry, 0, 180, 20
    PtPoly
    Rect 0, 0 - W, L, 0                          ' supa (teshikli)
    rr = Min2(L / nx, W / ny) * 0.3
    For i = 0 To nx - 1
        For j = 0 To ny - 1
            CutCircle L * (i + 0.5) / nx, 0 - W * (j + 0.5) / ny, rr
        Next j
    Next i
    Rect 0, 0 - W - H, L, 0 - W                  ' old devor
    Rect 0, 0 - 2 * W - H, L, 0 - W - H          ' tag
    Rect 0, 0 - 2 * W - 2 * H, L, 0 - 2 * W - H  ' orqa ichki devor (yelimlanadi)
    Rect 0 - H, 0 - W, 0, 0                      ' yon devorlar
    Rect L, 0 - W, L + H, 0
    SetRoot L / 2, hb / 2, True
End Sub

' ================================================================ 3-bosqich: yangi quruvchilar

' Ikki qiyali tomli quti: taglik L x W, old/orqa devor H, old devorga ikki tom qiyaligi (balandligi R) va til,
' yon devorlar beshburchak (tom qiyaliklariga yelim qanoti), quloqlar old/orqa devorda
Private Sub Roof(ByVal L As Double, ByVal W As Double, ByVal H As Double, ByVal R As Double)
    Dim sl As Double, th As Double, tk As Double, rr As Double, y As Double, a As Long, x0 As Double, sx As Double
    Dim eLen As Double, g As Double
    sl = Sqr(R * R + W * W / 4): th = Atn(R / (W / 2)) * 180 / PI
    tk = Min2(H * 0.6, 30): eLen = Min2(W * 0.4, H)
    Rect 0, 0, L, W                              ' taglik
    Rect 0, 0 - H, L, 0                          ' old devor
    Rect 0, W, L, W + H                          ' orqa devor
    Rect 0, 0 - H - sl, L, 0 - H                 ' tom: old qiyalik
    Rect 0, 0 - H - 2 * sl, L, 0 - H - sl        ' tom: orqa qiyalik
    y = 0 - H - 2 * sl: rr = Min2(tk * 0.6, L / 4)
    PtBegin                                      ' til (orqa devor ichiga kiradi)
    Pt L, y
    ArcPt L - rr, y - tk + rr, rr, 0, -90
    ArcPt rr, y - tk + rr, rr, 270, 180
    Pt 0, y
    PtPoly
    g = Min2(15, sl / 4)
    For a = 0 To 1
        If a = 0 Then x0 = 0: sx = -1 Else x0 = L: sx = 1
        PtBegin                                  ' yon devor (beshburchak)
        Pt x0, 0: Pt x0 + sx * H, 0: Pt x0 + sx * (H + R), W / 2: Pt x0 + sx * H, W: Pt x0, W
        PtPoly
        If a = 0 Then
            EdgeFlap x0 - H - R, W / 2, x0 - H, 0, g, g
            EdgeFlap x0 - H, W, x0 - H - R, W / 2, g, g
        Else
            EdgeFlap x0 + H, 0, x0 + H + R, W / 2, g, g
            EdgeFlap x0 + H + R, W / 2, x0 + H, W, g, g
        End If
        Ear x0, 0, sx, -1, eLen, H - 1, ""
        Ear x0, W, sx, 1, eLen, H - 1, ""
    Next a
    SetRoot L / 2, W / 2, False
    SetFold 0, 0 - H, L, 0 - H, 90 - th
    SetFold 0, 0 - H - sl, L, 0 - H - sl, 2 * th
    SetFold 0, y, L, y, 90 - th
End Sub

' Sumka-quti: taglik L x W, old/orqa devor qavariq (balandligi H), tepasida ruchka (ht), yon devorlar past
Private Sub Purse(ByVal L As Double, ByVal W As Double, ByVal H As Double, ByVal ht As Double)
    Dim b As Double, wt As Double, k As Long, i As Long, t As Double, y0 As Double, sy As Double
    Dim sd As Double, rr As Double, yt As Double
    b = W * 0.35: wt = Min2(L * 0.45, 90): sd = H * 0.55
    Rect 0, 0, L, W
    For k = 0 To 1
        If k = 0 Then y0 = 0: sy = -1 Else y0 = W: sy = 1
        yt = y0 + sy * H: rr = Min2(ht * 0.45, wt / 4)
        PtBegin
        For i = 0 To 12                          ' chap qirra
            t = i / 12
            Pt 0 - b * Sin(PI * t) + (L / 2 - wt / 2) * t * t, y0 + sy * H * t
        Next i
        If sy > 0 Then                           ' ruchka
            ArcPt L / 2 - wt / 2 + rr, yt + ht - rr, rr, 180, 90
            ArcPt L / 2 + wt / 2 - rr, yt + ht - rr, rr, 90, 0
        Else
            ArcPt L / 2 - wt / 2 + rr, yt - ht + rr, rr, 180, 270
            ArcPt L / 2 + wt / 2 - rr, yt - ht + rr, rr, 270, 360
        End If
        For i = 12 To 0 Step -1                  ' o'ng qirra
            t = i / 12
            Pt L + b * Sin(PI * t) - (L / 2 - wt / 2) * t * t, y0 + sy * H * t
        Next i
        PtPoly
        CutStadium L / 2, yt + sy * ht * 0.5, wt * 0.6, Min2(16, ht * 0.4)
    Next k
    Rect 0 - sd, 0, 0, W                          ' yon devorlar va tillari
    Rect L, 0, L + sd, W
    EdgeFlap 0 - sd, W, 0 - sd, 0, sd * 0.5, sd * 0.2
    EdgeFlap L + sd, 0, L + sd, W, sd * 0.5, sd * 0.2
    ' 3D: sodda model
    Box3 0, 0, 0, L, sd, W, False, True, True
    Face 0, 0, 0, L, 0, 0, L / 2 + wt / 2, H, W / 2, L / 2 - wt / 2, H, W / 2
    Face L, 0, W, 0, 0, W, L / 2 - wt / 2, H, W / 2, L / 2 + wt / 2, H, W / 2
    Face L / 2 - wt / 2, H, W / 2, L / 2 + wt / 2, H, W / 2, L / 2 + wt / 2, H + ht, W / 2, L / 2 - wt / 2, H + ht, W / 2
End Sub

' Yulduz quti: besh uchli yulduz (tashqi diametri dia), yon tasma (eni H) yelim tillari bilan,
' ikkita yulduz qopqoq alohida
Private Sub Star(ByVal dia As Double, ByVal H As Double, ByVal g As Double)
    Dim R As Double, ri As Double, k As Long, j As Long, e As Double, t As Double, b As Double, s As Double
    Dim sx(0 To 9) As Double, sz(0 To 9) As Double, x As Double, cy As Double
    R = dia / 2: ri = R * 0.382
    For k = 0 To 9
        If k Mod 2 = 0 Then t = R Else t = ri
        sx(k) = t * Cos(PI / 2 + k * PI / 5): sz(k) = t * Sin(PI / 2 + k * PI / 5)
    Next k
    e = Sqr((sx(1) - sx(0)) ^ 2 + (sz(1) - sz(0)) ^ 2)
    Glue 0, H, g
    t = Min2(8, e / 4)
    For k = 0 To 9
        x = k * e
        Rect x, 0, x + e, H
        For j = 0 To 1
            If j = 0 Then b = H: s = 1 Else b = 0: s = -1
            Poly x + 1, b, x + e - 1, b, x + e - 1 - t * 0.6, b + s * t, x + 1 + t * 0.6, b + s * t
        Next j
    Next k
    cy = 0 - t - 10 - R
    For j = 0 To 1                               ' qopqoqlar
        PtBegin
        For k = 0 To 9: Pt sx(k) + R + j * (2 * R + 10), sz(k) + cy: Next k
        PtPoly
    Next j
    ' 3D: yulduz prizma
    For k = 0 To 9
        j = (k + 1) Mod 10
        Face sx(k), 0, sz(k), sx(j), 0, sz(j), sx(j), H, sz(j), sx(k), H, sz(k)
    Next k
    For b = 0 To H Step H
        Face sx(1), b, sz(1), sx(3), b, sz(3), sx(5), b, sz(5), sx(7), b, sz(7), sx(9), b, sz(9)
        For k = 0 To 8 Step 2
            Face sx((k + 9) Mod 10), b, sz((k + 9) Mod 10), sx(k), b, sz(k), sx(k + 1), b, sz(k + 1)
        Next k
    Next b
End Sub

' n qirrali quti, ikki uchida o'tkir gulbarglar (yopiladi).
' fx: V - gulbarg to'g'ri uchburchak, B - tagi ko'pburchak (0-panelga ulangan, qolganlarida quloq)
Private Sub PetalN(ByVal n As Long, ByVal sd As Double, ByVal H As Double, ByVal g As Double, ByVal pet As Double, _
                   ByVal fx As String)
    Dim k As Long, i As Long, xa As Double, xb As Double, t As Double, j As Long, b As Double, s As Double
    Dim dh As Double
    If n < 3 Then n = 3
    Glue 0, H, g
    For k = 0 To n - 1
        xa = k * sd: xb = xa + sd
        Rect xa, 0, xb, H
        For j = 0 To 1
            If j = 0 Then b = H: s = 1 Else b = 0: s = -1
            If j = 1 And Has(fx, "B") Then
                If k > 0 Then                    ' tag quloqlari
                    dh = Min2(sd * 0.3, 15)
                    Poly xa, 0, xa + dh * 0.8, 0 - dh, xb - dh * 0.8, 0 - dh, xb, 0
                End If
            ElseIf Has(fx, "V") Then
                Poly xa, b, (xa + xb) / 2, b + s * pet, xb, b
            Else
                PtBegin
                For i = 0 To 10
                    t = i / 10
                    Pt xa + sd / 2 * (1 - Sqr(1 - t * t)), b + s * pet * t
                Next i
                For i = 9 To 0 Step -1
                    t = i / 10
                    Pt xb - sd / 2 * (1 - Sqr(1 - t * t)), b + s * pet * t
                Next i
                PtPoly
            End If
        Next j
    Next k
    If Has(fx, "B") Then                         ' ko'pburchak tag
        RegPoly sd, 0, 0, 0, n
        PtPoly
    End If
    SetRoot sd / 2, H / 2, True
    For k = 1 To n - 1: SetFold k * sd, 0, k * sd, H, 360 / n: Next k
    SetFold 0, 0, 0, H, 360 / n
    If Has(fx, "V") Then                         ' uchburchak gulbarglar tepada uchrashadi
        t = sd / (2 * Tan(PI / n))
        t = Atn(Sqr(Max2(pet * pet - t * t, 1)) / t) * 180 / PI
        For k = 0 To n - 1: SetFold k * sd, H, (k + 1) * sd, H, 90 - t: Next k
    End If
    SetLid 0, H, n * sd, H
End Sub
